import requests

BASE_URL = "https://wafus.vercel.app"

# 1. Fetch Random Waifu (Gacha Roll)
random_waifu = requests.get(f"{BASE_URL}/Random").json()
print("Random Waifu:", random_waifu["data"]["name"])
print("Rarity:", random_waifu["data"]["rarity"])
print("Image URL:", random_waifu["data"]["img_url"])

# 2. Search Waifu by Name
res = requests.get(f"{BASE_URL}/Find", params={"name": "Komi"}).json()
print("Search Result:", res)
// Using native fetch
const BASE_URL = "https://wafus.vercel.app";

// 1. Fetch Random Waifu
async function getRandomWaifu() {
  const res = await fetch(`${BASE_URL}/Random`);
  const data = await res.json();
  console.log(`Waifu: ${data.data.name} (${data.data.rarity})`);
  console.log(`Image: ${data.data.img_url}`);
  return data.data;
}

// 2. Search by Character Name
async function searchWaifu(name) {
  const res = await fetch(`${BASE_URL}/Find?name=${encodeURIComponent(name)}`);
  return await res.json();
}

getRandomWaifu();
package main

import (
	"encoding/json"
	"fmt"
	"net/http"
)

type WaifuResponse struct {
	Status string `json:"status"`
	Data   struct {
		Name   string `json:"name"`
		Rarity string `json:"rarity"`
		ImgURL string `json:"img_url"`
		ID     string `json:"id"`
	} `json:"data"`
}

func main() {
	resp, err := http.Get("https://wafus.vercel.app/Random")
	if err != nil {
		panic(err)
	}
	defer resp.Body.Close()

	var result WaifuResponse
	json.NewDecoder(resp.Body).Decode(&result)

	fmt.Printf("Waifu: %s [%s]\nURL: %s\n", result.Data.Name, result.Data.Rarity, result.Data.ImgURL)
}
